# ---------------------------------------------------------------------------
# Slapper — physical-modelling funk bass (a string, its attacks, an amp)
#   SlapperCode  INTERFACE target (plugin, sl_tests)
#   Slapper      the VST3 / AU (aumu) / Standalone instrument
# ---------------------------------------------------------------------------

juce_add_binary_data(SlapperPresets
    HEADER_NAME SlapperPresets.h
    NAMESPACE   SLPresets
    SOURCES
        Presets/thumb_party.json
        Presets/pop_shop.json
        Presets/finger_round.json
        Presets/ghost_groove.json
        Presets/dusty_wood.json
        Presets/disco_octaves.json
        Presets/soul_mute.json
        Presets/slap_attack.json
        Presets/mothership.json
        Presets/rubber_band.json
        Presets/floor_pedal.json
        Presets/solid_fm.json
        Presets/fm_slap.json
        Presets/glass_bass.json
        Presets/boogie_pulse.json
        Presets/mirrorball.json)
set_target_properties(SlapperPresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Pluck's poses (generated by scripts/make_pluck_frames.sh)
file(GLOB SL_PLUCK_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/pluck_*.png)
juce_add_binary_data(SlapperArt
    HEADER_NAME SlapperArt.h
    NAMESPACE   SLArt
    SOURCES     ${SL_PLUCK_FRAMES})
set_target_properties(SlapperArt PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(SlapperCode INTERFACE)
target_sources(SlapperCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SlapperVoice.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SlapperAmp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SynthVoice.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SlapperDsp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SlapperProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SlapperEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/PluckMascot.cpp)
target_include_directories(SlapperCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(SlapperCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    SlapperPresets
    SlapperArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(SlapperCode INTERFACE
    SL_PRODUCT_NAME="${SL_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    SL_VERSION_STRING="${SL_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_sl_copy TRUE)
else()
    set(_sl_copy FALSE)
endif()

juce_add_plugin(Slapper
    PRODUCT_NAME                "${SL_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${SL_BUNDLE_ID}"
    VERSION                     "${SL_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${SL_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    IS_SYNTH                    TRUE
    NEEDS_MIDI_INPUT            TRUE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    HARDENED_RUNTIME_ENABLED    TRUE
    AU_MAIN_TYPE                kAudioUnitType_MusicDevice
    VST3_CATEGORIES             Instrument Synth
    DESCRIPTION                 "One-knob physical-modelling funk bass (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_sl_copy})

target_sources(Slapper PRIVATE Source/PluginEntry.cpp)
target_link_libraries(Slapper PRIVATE
    SlapperCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(Slapper)
