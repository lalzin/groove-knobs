# ---------------------------------------------------------------------------
# DAFT CHUNK — an instrument: slices a sample on its transients, plays the
# slices in MIDI, and makes loops (CHOP: how wild the slicing gets)
#   DaftChunkCode  INTERFACE target (plugin, dc_tests)
#   DaftChunk      the VST3 / AU / Standalone instrument
# ---------------------------------------------------------------------------

juce_add_binary_data(DaftChunkPresets
    HEADER_NAME DaftChunkPresets.h
    NAMESPACE   DCPresets
    SOURCES     Presets/default.json)
set_target_properties(DaftChunkPresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Chunky's poses (generated by scripts/make_chunky_frames.sh)
file(GLOB DC_CHUNKY_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/chunky_*.png)
gk_add_art(DaftChunkArt
    HEADER_NAME DaftChunkArt.h
    NAMESPACE   DCArt
    SOURCES     ${DC_CHUNKY_FRAMES})

add_library(DaftChunkCode INTERFACE)
target_sources(DaftChunkCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/Analysis.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SampleLoader.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/DaftChunkEngine.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/DaftChunkProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/DaftChunkEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/WaveformView.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/ChunkyMascot.cpp)
target_include_directories(DaftChunkCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(DaftChunkCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    DaftChunkPresets
    DaftChunkArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(DaftChunkCode INTERFACE
    DC_PRODUCT_NAME="${DC_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    DC_VERSION_STRING="${DC_VERSION}"
    JUCE_USE_MP3AUDIOFORMAT=1   # MP3 everywhere (Windows has no system decoder in JUCE)
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_dc_copy TRUE)
else()
    set(_dc_copy FALSE)
endif()

juce_add_plugin(DaftChunk
    PRODUCT_NAME                "${DC_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${DC_BUNDLE_ID}"
    VERSION                     "${DC_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${DC_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    ICON_BIG                    ${CMAKE_CURRENT_SOURCE_DIR}/Assets/app_icon.png  # Chunky
    IS_SYNTH                    TRUE
    NEEDS_MIDI_INPUT            TRUE
    NEEDS_MIDI_OUTPUT           FALSE  # step 3: TRUE with MIDI CAPTURE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    HARDENED_RUNTIME_ENABLED    TRUE
    AU_MAIN_TYPE                kAudioUnitType_MusicDevice
    VST3_CATEGORIES             Instrument Sampler
    DESCRIPTION                 "Slices a sample, plays it in MIDI, makes loops (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_dc_copy})

target_sources(DaftChunk PRIVATE Source/PluginEntry.cpp)
target_link_libraries(DaftChunk PRIVATE
    DaftChunkCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(DaftChunk)
gk_bundle_art(DaftChunk DaftChunkArt)
