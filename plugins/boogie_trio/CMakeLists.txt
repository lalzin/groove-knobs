# ---------------------------------------------------------------------------
# Boogie Trio — 70s string-machine triple chorus (second Groove Knobs plugin)
#   BoogieTrioCode  INTERFACE target (plugin, bt_render, bt_tests)
#   BoogieTrio      the VST3 / AU / Standalone plugin
# ---------------------------------------------------------------------------

juce_add_binary_data(BoogieTrioPresets
    HEADER_NAME BoogieTrioPresets.h
    NAMESPACE   BTPresets
    SOURCES
        Presets/disco_strings.json
        Presets/lush_pad.json
        Presets/warble.json
        Presets/velvet.json
        Presets/funk_keys.json
        Presets/jet_set.json
        Presets/disco_gate.json
        Presets/moon_phaser.json)
set_target_properties(BoogieTrioPresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Velvet's poses (generated by scripts/make_velvet_frames.sh)
file(GLOB BT_VELVET_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/velvet_*.png)
juce_add_binary_data(BoogieTrioArt
    HEADER_NAME BoogieTrioArt.h
    NAMESPACE   BTArt
    SOURCES     ${BT_VELVET_FRAMES})
set_target_properties(BoogieTrioArt PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(BoogieTrioCode INTERFACE)
target_sources(BoogieTrioCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BoogieTrioDsp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BoogieTrioProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BoogieTrioEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/VelvetMascot.cpp)
target_include_directories(BoogieTrioCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(BoogieTrioCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    BoogieTrioPresets
    BoogieTrioArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(BoogieTrioCode INTERFACE
    BT_PRODUCT_NAME="${BT_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    BT_VERSION_STRING="${BT_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_bt_copy TRUE)
else()
    set(_bt_copy FALSE)
endif()

juce_add_plugin(BoogieTrio
    PRODUCT_NAME                "${BT_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${BT_BUNDLE_ID}"
    VERSION                     "${BT_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${BT_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    # Standalone app: audio input under the hardened runtime (signed builds)
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT  "The standalone app processes your audio input."
    HARDENED_RUNTIME_ENABLED    TRUE
    HARDENED_RUNTIME_OPTIONS    com.apple.security.device.audio-input
    AU_MAIN_TYPE                kAudioUnitType_Effect
    VST3_CATEGORIES             Fx Modulation
    DESCRIPTION                 "One-knob 70s string-machine chorus (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_bt_copy})

target_sources(BoogieTrio PRIVATE Source/PluginEntry.cpp)
target_link_libraries(BoogieTrio PRIVATE
    BoogieTrioCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(BoogieTrio)
