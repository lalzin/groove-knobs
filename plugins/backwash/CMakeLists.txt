# ---------------------------------------------------------------------------
# Backwash — tempo-synced reverse reverb with 2 s lookahead (third Groove Knobs plugin)
#   BackwashCode  INTERFACE target (plugin, bt_render, bt_tests)
#   Backwash      the VST3 / AU / Standalone plugin
# ---------------------------------------------------------------------------

juce_add_binary_data(BackwashPresets
    HEADER_NAME BackwashPresets.h
    NAMESPACE   BWPresets
    SOURCES
        Presets/snare_swell.json
        Presets/vocal_suck.json
        Presets/clap_rewind.json
        Presets/ghost_pad.json
        Presets/big_inhale.json
        Presets/reverse_echo.json
        Presets/stutter_suck.json
        Presets/dusty_tape.json)
set_target_properties(BackwashPresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Swell's poses (generated by scripts/make_swell_frames.sh)
file(GLOB BW_SWELL_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/swell_*.png)
juce_add_binary_data(BackwashArt
    HEADER_NAME BackwashArt.h
    NAMESPACE   BWArt
    SOURCES     ${BW_SWELL_FRAMES})
set_target_properties(BackwashArt PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(BackwashCode INTERFACE)
target_sources(BackwashCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BackwashDsp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BackwashProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BackwashEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/SwellMascot.cpp)
target_include_directories(BackwashCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(BackwashCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    BackwashPresets
    BackwashArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(BackwashCode INTERFACE
    BW_PRODUCT_NAME="${BW_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    BW_VERSION_STRING="${BW_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_bt_copy TRUE)
else()
    set(_bt_copy FALSE)
endif()

juce_add_plugin(Backwash
    PRODUCT_NAME                "${BW_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${BW_BUNDLE_ID}"
    VERSION                     "${BW_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${BW_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    ICON_BIG                    ${CMAKE_CURRENT_SOURCE_DIR}/Assets/app_icon.png  # the mascot (app icon, macOS and Windows)
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    # Standalone app: audio input under the hardened runtime (signed builds)
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT  "The standalone app processes your audio input."
    HARDENED_RUNTIME_ENABLED    TRUE
    HARDENED_RUNTIME_OPTIONS    com.apple.security.device.audio-input
    AU_MAIN_TYPE                kAudioUnitType_Effect
    VST3_CATEGORIES             Fx Reverb
    DESCRIPTION                 "One-knob tempo-synced reverse reverb (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_bt_copy})

target_sources(Backwash PRIVATE Source/PluginEntry.cpp)
target_link_libraries(Backwash PRIVATE
    BackwashCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(Backwash)
