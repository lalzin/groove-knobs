# ---------------------------------------------------------------------------
# BALLROOM — an algorithmic reverb, SPACE as the one knob (early + late)
#   BallroomCode  INTERFACE target (plugin, tests)
#   Ballroom      the VST3 / AU / Standalone effect
# ---------------------------------------------------------------------------

juce_add_binary_data(BallroomPresets
    HEADER_NAME BallroomPresets.h
    NAMESPACE   BLPresets
    SOURCES
        Presets/disco_ballroom.json
        Presets/tight_club.json
        Presets/drum_room.json
        Presets/snare_chamber.json
        Presets/gated_80s.json
        Presets/vocal_plate.json
        Presets/mirror_ball.json
        Presets/warm_hall.json
        Presets/cathedral.json
        Presets/velvet_pad.json
        Presets/afterparty.json
        Presets/frozen_floor.json
        Presets/clear_vocal.json
        Presets/power_snare.json
        Presets/disco_strings.json
        Presets/funk_room.json
        Presets/glitter_plate.json
        Presets/dub_chamber.json
        Presets/slapback_club.json
        Presets/chrome_room.json
        Presets/breathing_sky.json
        Presets/tiny_booth.json)
set_target_properties(BallroomPresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Mira's poses (generated by scripts/make_mira_frames.sh)
file(GLOB BL_MIRA_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/mira_*.png)
juce_add_binary_data(BallroomArt
    HEADER_NAME BallroomArt.h
    NAMESPACE   BLArt
    SOURCES     ${BL_MIRA_FRAMES})
set_target_properties(BallroomArt PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(BallroomCode INTERFACE)
target_sources(BallroomCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/LateFdn.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/EarlyNet.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/Ballroom.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BallroomDsp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BallroomProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/BallroomEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/MiraMascot.cpp)
target_include_directories(BallroomCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(BallroomCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    BallroomPresets
    BallroomArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(BallroomCode INTERFACE
    BL_PRODUCT_NAME="${BL_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    BL_VERSION_STRING="${BL_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_bl_copy TRUE)
else()
    set(_bl_copy FALSE)
endif()

juce_add_plugin(Ballroom
    PRODUCT_NAME                "${BL_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${BL_BUNDLE_ID}"
    VERSION                     "${BL_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${BL_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT  "The standalone app processes your audio input."
    HARDENED_RUNTIME_ENABLED    TRUE
    HARDENED_RUNTIME_OPTIONS    com.apple.security.device.audio-input
    AU_MAIN_TYPE                kAudioUnitType_Effect
    VST3_CATEGORIES             Fx Reverb
    DESCRIPTION                 "An algorithmic reverb, SPACE in one knob (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_bl_copy})

target_sources(Ballroom PRIVATE Source/PluginEntry.cpp)
target_link_libraries(Ballroom PRIVATE
    BallroomCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(Ballroom)
