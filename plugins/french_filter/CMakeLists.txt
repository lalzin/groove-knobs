# ---------------------------------------------------------------------------
# French Filter
#
#   FrenchFilterCode  INTERFACE target with all plugin sources + groove_core.
#                     Linked by the plugin, ff_render and the tests (each
#                     consumer compiles it, as JUCE modules are INTERFACE too).
#   FrenchFilter      the VST3 / AU / Standalone plugin.
# ---------------------------------------------------------------------------

juce_add_binary_data(FrenchFilterPresets
    HEADER_NAME FrenchFilterPresets.h
    NAMESPACE   FFPresets
    SOURCES
        Presets/filtered_disco.json
        Presets/sunrise_sweep.json
        Presets/muffled_club.json
        Presets/acid_disco.json
        Presets/touch_98.json
        Presets/robot_funk.json
        Presets/sounds_better.json
        Presets/lady_groove.json
        Presets/space.json
        Presets/clean_cut.json
        Presets/sub_diver.json
        Presets/dub_wash.json
        Presets/stereo_swirl.json
        Presets/hard_pump.json
        Presets/crunch_loop.json
        Presets/laser_riser.json)
set_target_properties(FrenchFilterPresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Mascot poses (generated by scripts/make_mascot_frames.sh)
file(GLOB FF_MASCOT_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/mascot_*.png)
juce_add_binary_data(FrenchFilterArt
    HEADER_NAME FrenchFilterArt.h
    NAMESPACE   FFArt
    SOURCES     ${FF_MASCOT_FRAMES})
set_target_properties(FrenchFilterArt PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(FrenchFilterCode INTERFACE)
target_sources(FrenchFilterCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/FrenchFilterDsp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/FrenchFilterProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/FrenchFilterEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/FrenchFilterMascot.cpp)
target_include_directories(FrenchFilterCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(FrenchFilterCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    FrenchFilterPresets
    FrenchFilterArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(FrenchFilterCode INTERFACE
    FF_PRODUCT_NAME="${FF_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    FF_VERSION_STRING="${FF_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_ff_copy TRUE)
else()
    set(_ff_copy FALSE)
endif()

juce_add_plugin(FrenchFilter
    PRODUCT_NAME                "${FF_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${FF_BUNDLE_ID}"
    VERSION                     "${FF_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${FF_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    ICON_BIG                    ${CMAKE_CURRENT_SOURCE_DIR}/Assets/app_icon.png  # the mascot (app icon, macOS and Windows)
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    # Standalone app: audio input under the hardened runtime (signed builds)
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT  "The standalone app processes your audio input."
    HARDENED_RUNTIME_ENABLED    TRUE
    HARDENED_RUNTIME_OPTIONS    com.apple.security.device.audio-input
    AU_MAIN_TYPE                kAudioUnitType_Effect
    VST3_CATEGORIES             Fx Filter
    DESCRIPTION                 "One-knob French Touch filter (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_ff_copy})

target_sources(FrenchFilter PRIVATE Source/PluginEntry.cpp)
target_link_libraries(FrenchFilter PRIVATE
    FrenchFilterCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(FrenchFilter)
