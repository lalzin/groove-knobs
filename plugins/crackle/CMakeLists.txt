# ---------------------------------------------------------------------------
# Crackle — one-knob lo-fi grain (vinyl, tape, radio, bitcrush)
#   CrackleCode  INTERFACE target (plugin, cr_render, cr_tests)
#   Crackle      the VST3 / AU / Standalone effect
# ---------------------------------------------------------------------------

juce_add_binary_data(CracklePresets
    HEADER_NAME CracklePresets.h
    NAMESPACE   CRPresets
    SOURCES
        Presets/vinyl_sunday.json
        Presets/cassette_dream.json
        Presets/am_radio_ghost.json
        Presets/basement_tape.json
        Presets/dusty_drum_loop.json
        Presets/walkman_bus.json
        Presets/broken_sampler.json
        Presets/sunday_kitchen.json
        Presets/telephone_booth.json
        Presets/crushed_dub_tape.json)
set_target_properties(CracklePresets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Hiss's poses (generated by scripts/make_hiss_frames.sh)
file(GLOB CR_HISS_FRAMES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Assets/hiss_*.png)
juce_add_binary_data(CrackleArt
    HEADER_NAME CrackleArt.h
    NAMESPACE   CRArt
    SOURCES     ${CR_HISS_FRAMES})
set_target_properties(CrackleArt PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(CrackleCode INTERFACE)
target_sources(CrackleCode INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/CrackleChain.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/Saturator.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/CrackleDsp.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/CrackleProcessor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/CrackleEditor.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/HissMascot.cpp)
target_include_directories(CrackleCode INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(CrackleCode INTERFACE
    groove::groove_core
    GrooveCoreAssets
    CracklePresets
    CrackleArt
    juce::juce_audio_utils
    juce::juce_dsp)
target_compile_definitions(CrackleCode INTERFACE
    CR_PRODUCT_NAME="${CR_PRODUCT_NAME}"
    GK_SUITE_NAME="${GK_SUITE_NAME}"
    GK_COMPANY_NAME="${GK_COMPANY_NAME}"
    CR_VERSION_STRING="${CR_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_MODAL_LOOPS_PERMITTED=0)

if(GK_COPY_PLUGINS)
    set(_cr_copy TRUE)
else()
    set(_cr_copy FALSE)
endif()

juce_add_plugin(Crackle
    PRODUCT_NAME                "${CR_PRODUCT_NAME}"
    COMPANY_NAME                "${GK_COMPANY_NAME}"
    BUNDLE_ID                   "${CR_BUNDLE_ID}"
    VERSION                     "${CR_VERSION}"
    PLUGIN_MANUFACTURER_CODE    ${GK_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${CR_PLUGIN_CODE}
    FORMATS                     VST3 AU Standalone
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT  "The standalone app processes your audio input."
    HARDENED_RUNTIME_ENABLED    TRUE
    HARDENED_RUNTIME_OPTIONS    com.apple.security.device.audio-input
    AU_MAIN_TYPE                kAudioUnitType_Effect
    VST3_CATEGORIES             Fx Distortion
    DESCRIPTION                 "One-knob lo-fi grain: vinyl, tape, radio, bitcrush (${GK_SUITE_NAME})"
    COPY_PLUGIN_AFTER_BUILD     ${_cr_copy})

target_sources(Crackle PRIVATE Source/PluginEntry.cpp)
target_link_libraries(Crackle PRIVATE
    CrackleCode
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags)
gk_enable_warnings(Crackle)
